    def to_permissions(perms, apply_mask):
        if apply_mask and mask_perms is not None:
            perms = "".join(c for c in perms if c in mask_perms)
        words = []
        if "r" in perms:
            words.append("read")
        if "w" in perms:
            words.append("write")
        if "x" in perms:
            words.append("execute")
        return words

    entries = []
    for entity_type, entity, perms in raw_entries:
        if entity_type == "user" and entity == "":

            # users and the group class with the mask, never the owner. The mask
            # test itself is character membership, which works because getfacl
